cmake_minimum_required(VERSION 3.16)
project(gazebo_start_order_tests LANGUAGES CXX)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs)
find_package(Threads REQUIRED)
set(scene_dir "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(xgc2_simulation_ros_master SHARED "${scene_dir}/src/ros_master_registration.cpp")
target_compile_features(xgc2_simulation_ros_master PUBLIC cxx_std_17)
target_compile_options(xgc2_simulation_ros_master PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(xgc2_simulation_ros_master PUBLIC "${scene_dir}/include")
target_include_directories(xgc2_simulation_ros_master SYSTEM PUBLIC ${catkin_INCLUDE_DIRS})
target_link_libraries(xgc2_simulation_ros_master PRIVATE Threads::Threads ${catkin_LIBRARIES})
add_executable(ros_master_fixture ros_master_fixture.cpp)
target_compile_features(ros_master_fixture PRIVATE cxx_std_17)
target_compile_options(ros_master_fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(ros_master_fixture PRIVATE xgc2_simulation_ros_master ${catkin_LIBRARIES})
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME parallel_start COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../test_parallel_start.py"
  --fixture $<TARGET_FILE:ros_master_fixture>)
add_test(NAME launch_start_order COMMAND ${Python3_EXECUTABLE}
  "${CMAKE_CURRENT_SOURCE_DIR}/../test_launch_start_order.py")
add_test(NAME peer_down_long COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../test_parallel_start.py"
  --fixture $<TARGET_FILE:ros_master_fixture> --long)
set_tests_properties(peer_down_long PROPERTIES LABELS "long" TIMEOUT 200)
